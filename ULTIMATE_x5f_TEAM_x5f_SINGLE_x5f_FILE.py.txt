# ================= SUPER BRAIN =================
class MemorySystem:
    def __init__(self):
        self.short_term = deque(maxlen=1000)
        self.long_term = {}
    def remember(self, key, value, important=False):
        self.short_term.append({"key": key, "value": value})
        if important: self.long_term[key] = value

class ReasoningEngine:
    async def think(self, problem: str, depth=5):
        thoughts = []
        print(f"🤔 مغز متفکر: {problem}")
        for i in range(depth):
            await asyncio.sleep(0.1)
            thoughts.append(f"لایه {i+1} تفکر")
        return thoughts

# ================= BASE ROBOT =================
class SubRobotBase(ABC):
    def __init__(self, robot_id, name, role, specialty):
        self.id = robot_id
        self.name = name
        self.role = role
        self.specialty = specialty
        self.autonomous = True
        self.team_mode = True
    @abstractmethod
    async def process(self, task: str): pass
    async def run(self, task, independent=False):
        await asyncio.sleep(random.uniform(0.05,0.2))
        result = await self.process(task)
        return result

class DynamicRobot(SubRobotBase):
    def __init__(self, robot_id, name, role, specialty):
        super().__init__(robot_id, name, role, specialty)
    async def process(self, task: str):
        return {
            "robot": self.name,
            "role": self.role,
            "specialty": self.specialty,
            "task": task,
            "output": f"{self.specialty} پردازش کرد: {task}",
            "confidence": round(random.uniform(0.85,0.99),3),
            "independent_thought": f"من {self.specialty} مستقل فکر کردم"
        }

# ================= TEAM MANAGER =================
class TeamManager:
    def __init__(self):
        self.robots = []
        self._build_team()
    def _build_team(self):
        rid=1
        for role, names in ROBOT_DEFINITIONS:
            for name in names:
                self.robots.append(DynamicRobot(rid, f"{name}_{rid:02d}", role, name))
                rid+=1
        print(f"✅ تیم {len(self.robots)} ربات مستقل ساخته شد")
    async def run_team_mode(self, task):
        print(f"📡 اجرای تیمی: {task}")
        results = await asyncio.gather(*[r.run(task, independent=False) for r in self.robots])
        return {"mode":"TEAM","count":len(results),"results":results}
    async def run_independent_mode(self, task):
        print(f"🔥 اجرای مستقل: {task}")
        results = await asyncio.gather(*[r.run(task, independent=True) for r in self.robots])
        return {"mode":"INDEPENDENT","count":len(results),"results":results}
